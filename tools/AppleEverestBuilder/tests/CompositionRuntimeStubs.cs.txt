    // The matcher and its sound/debris registration are verified separately
    // against pinned code. This adapter observes graphics lifecycle ordering.
    public sealed class Autotiler
    {
        public readonly string Path;
        public readonly HashSet<char> TileIds=new();
        public static readonly List<string> Constructions=new();
        public Autotiler(string path)
        {
            Path=path;Constructions.Add(path);
            foreach(XmlElement tile in Monocle.Calc.LoadContentXML(path).GetElementsByTagName("Tileset"))
            {
                TileIds.Add(tile.GetAttribute("id")[0]);
                if(tile.HasAttribute("sound"))SurfaceIndex.TileToIndex[tile.GetAttribute("id")[0]]=int.Parse(tile.GetAttribute("sound"));
            }
        }
    }

    internal static class Everest
    {
        internal static class Events
        {
            internal static class Level
            {
                internal delegate bool LoadEntityHandler(Celeste.Level level,Celeste.LevelData room,Microsoft.Xna.Framework.Vector2 offset,Celeste.EntityData data);
                internal static event LoadEntityHandler OnLoadEntity;
                internal static int HandlerCount=>OnLoadEntity?.GetInvocationList().Length??0;
                internal static bool LoadEntity(Celeste.Level level,Celeste.LevelData room,Microsoft.Xna.Framework.Vector2 offset,Celeste.EntityData data)
                {
                    if(OnLoadEntity==null)return false;
                    foreach(LoadEntityHandler handler in OnLoadEntity.GetInvocationList())if(handler(level,room,offset,data))return true;
                    return false;
                }
            }
        }
    }
