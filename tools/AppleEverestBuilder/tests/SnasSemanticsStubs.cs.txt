    internal static class Trace
    {
        internal static readonly List<string> Events = new();

    internal sealed class RecordedRandom(int seed) : Random(seed)
    {
        public override int Next(int maxValue)
        {
            int value = base.Next(maxValue);
            Trace.Add("rng:" + maxValue + ":" + value);
            return value;
        }
    }

    public struct Vector2(float x, float y)
    {
        public float X = x, Y = y;
        public static Vector2 Zero => new(0,0);
        public static Vector2 One => new(1,1);
        public static Vector2 operator +(Vector2 a, Vector2 b) => new(a.X+b.X,a.Y+b.Y);
        public static Vector2 operator -(Vector2 a, Vector2 b) => new(a.X-b.X,a.Y-b.Y);
        public static Vector2 operator *(Vector2 a, float b) => new(a.X*b,a.Y*b);
        public override string ToString() => X.ToString(CultureInfo.InvariantCulture)+","+Y.ToString(CultureInfo.InvariantCulture);
    }

    public class Entity
    {
        public Vector2 Position;
        public float actualDepth;
        public int Depth { get => (int)actualDepth; set => actualDepth=value; }
        public bool Active=true,Visible=true,Collidable=true;
        public Scene Scene;
        public Collider Collider;
        public readonly ComponentList Components;
        public float Left=>Position.X+(Collider?.Left??0);
        public float Top=>Position.Y+(Collider?.Top??0);
        public float Right=>Left+(Collider?.Width??0);
        public float Bottom=>Top+(Collider?.Height??0);
        public float CenterX=>(Left+Right)/2;
        public float CenterY=>(Top+Bottom)/2;
        public Entity(Vector2 position=default) { Position=position;Components=new ComponentList(this); }
        public void Add(Component component) => Components.Add(component);
        public void Remove(Component component) => Components.Remove(component);
        public void RemoveSelf() { Conformance.Trace.Add("remove-request");Scene?.Entities.Remove(this); }
        public virtual void Added(Scene scene)
        { Scene=scene;foreach(Component c in Components)c.EntityAdded(scene); }
        public virtual void Awake(Scene scene) { foreach(Component c in Components)c.EntityAwake(); }
        public virtual void Removed(Scene scene)
        { foreach(Component c in Components)c.EntityRemoved(scene);Scene=null;Conformance.Trace.Add("removed"); }
        public virtual void Update() => Components.Update();
        public virtual void Render() { }
        public virtual void DebugRender(Camera camera) { }
        public virtual void HandleGraphicsReset() { }
        public virtual void HandleGraphicsCreate() { }
        public bool TagCheck(int tags)=>false;
        public bool TagFullCheck(int tags)=>false;
        public T SceneAs<T>() where T:Scene => Scene as T;
        public bool CollideCheck(Entity other)=>true;
    }
