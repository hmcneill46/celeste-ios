    private static object Get(object target,string name)
    {
        Type type=target as Type??target.GetType();object instance=target is Type?null:target;
        for(Type current=type;current!=null;current=current.BaseType)
        {
            FieldInfo field=current.GetField(name,All|BindingFlags.DeclaredOnly);
            if(field!=null)return field.GetValue(instance);
            PropertyInfo property=current.GetProperty(name,All|BindingFlags.DeclaredOnly);
            if(property!=null)return property.GetValue(instance);
        }
        throw new MissingMemberException(type.FullName,name);
    }
    private static void Set(object target,string name,object value)
    {
        Type type=target as Type??target.GetType();object instance=target is Type?null:target;
        for(Type current=type;current!=null;current=current.BaseType)
        {
            FieldInfo field=current.GetField(name,All|BindingFlags.DeclaredOnly);
            if(field!=null){field.SetValue(instance,value);return;}
            PropertyInfo property=current.GetProperty(name,All|BindingFlags.DeclaredOnly);
            if(property!=null){property.SetValue(instance,value);return;}
        }
        throw new MissingMemberException(type.FullName,name);
    }
    private static object Call(object target,string name,params object[] args)
    {
        Type type=target as Type??target.GetType();object instance=target is Type?null:target;
        try{return type.GetMethods(All).Single(m=>m.Name==name&&m.GetParameters().Length==args.Length).Invoke(instance,args);}
        catch(TargetInvocationException ex){System.Runtime.ExceptionServices.ExceptionDispatchInfo.Capture(ex.InnerException).Throw();throw;}
    }
    private static void Reject(Action action,string what,string expected="")
    {
        try{action();}catch(InvalidOperationException error) when(expected.Length==0||error.Message.Contains(expected,StringComparison.Ordinal)){checks++;return;}
        throw new Exception("unproved creation accepted: "+what);
    }

    {
        Atlas result=new();

            typeof(VirtualAsset).GetProperty("Width",All).SetValue(texture,sheet.GetProperty("width").GetInt32());
            typeof(VirtualAsset).GetProperty("Height",All).SetValue(texture,sheet.GetProperty("height").GetInt32());

            {
                string key=item.GetProperty("key").GetString();int[] v=item.GetProperty("rect").EnumerateArray().Select(x=>x.GetInt32()).ToArray();

            }

    private static Player NewPlayer(Vector2 at)
    {
        // The device Player constructor enters IOSTouchControls, which needs
        // Microsoft.iOS and UIKit. Supply only its host fixture state, then
        // execute the unchanged compiled collider/cassette methods and actual
        // StateMachine callbacks. This does not claim Player constructor or
        // full Player.Update execution on this host.
        Player player=(Player)RuntimeHelpers.GetUninitializedObject(typeof(Player));
        Actor basis=new(at);
        foreach(Type type in new[]{typeof(Entity),typeof(Actor)})
            foreach(FieldInfo field in type.GetFields(All|BindingFlags.DeclaredOnly).Where(f=>!f.IsStatic))
                field.SetValue(player,field.GetValue(basis));
        Set(player.Components,"Entity",player);
        player.Collider=new Hitbox(8,11,-4,-11);
        player.Sprite=new PlayerSprite(PlayerSpriteMode.Madeline);player.Add(player.Sprite);
        player.StateMachine=new StateMachine(26);player.Add(player.StateMachine);
        player.StateMachine.SetCallbacks(0,null,null,()=>Call(player,"NormalBegin"),()=>Call(player,"NormalEnd"));
        player.StateMachine.SetCallbacks(21,()=> (int)Call(player,"CassetteFlyUpdate"),
            ()=> (IEnumerator)Call(player,"CassetteFlyCoroutine"),()=>Call(player,"CassetteFlyBegin"),()=>Call(player,"CassetteFlyEnd"));
        player.StateMachine.State=0;Set(player,"triggersInside",new HashSet<Trigger>());
        return player;
    }

    // Host-only explicit iterator states transcribed from the pinned Max gate
    // (source c7ab870c8fa16ab93acf68cc42e65ca5164c27538abec39cb2d24befdaa0cdf2).
    // This oracle shares the *unchanged canonical* coroutine/tween scheduler;
    // it does not read candidate fields or call its sequence implementation.
    private sealed class PinnedGateOracle : Solid, IEnumerator
    {
        private int phase;
        private float remaining;
        private readonly Vector2 initial,destination;
        private readonly int mutation;
        public float Rate;
        public object Current { get; private set; }
        public PinnedGateOracle(Vector2 initial,Vector2 destination,int mutation) : base(initial,32,32,false)
        { this.initial=initial;this.destination=destination;this.mutation=mutation;Add(new Coroutine(this)); }
        public void Reset()=>throw new NotSupportedException();
        public bool MoveNext()
        {
            Current=null;
            switch(phase)
            {
                case 0: phase=1;Current=.1f;return true;
                case 1: phase=2;goto case 2;
                case 2:
                    if(Rate<1f){Rate+=Engine.DeltaTime/.5f;return true;}
                    phase=3;Current=.1f;return true;
                case 3:
                    Tween tween=Tween.Create(Tween.TweenMode.Oneshot,mutation==1?Ease.Linear:Ease.CubeOut,mutation==2?1.8f:2f,true);
                    tween.OnUpdate=value=>MoveTo(Vector2.Lerp(initial,destination,value.Eased));Add(tween);
                    if(mutation==4)tween.Update();
                    remaining=mutation==3?2f:1.8f;phase=4;return true;
                case 4:
                    remaining-=Engine.DeltaTime;if(remaining>0f)return true;
                    phase=5;goto case 5;
                case 5:
                    if(Rate>0f){Rate-=Engine.DeltaTime*4f;return true;}
                    Rate=0;phase=6;return false;
                default:return false;
            }
        }
    }

        bool Tween,float Duration,float TimeLeft,float Percent,float Eased);

            tween?.TimeLeft??0,tween?.Percent??0,tween?.Eased??0);
    }
